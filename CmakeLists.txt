cmake_minimum_required(VERSION 3.21)

project(travis VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)

include(CTest)

find_package(Qt6 REQUIRED COMPONENTS Core Sql Test)

add_library(travis_data_database
    src/app/app_context.cpp
    src/data/database/application_identity.cpp
    src/data/database/database_bootstrap.cpp
    src/data/database/database_manager.cpp
    src/data/database/database_paths.cpp
    src/data/repositories/asset_repository.cpp
    src/data/repositories/base_repository.cpp
    src/data/repositories/component_repository.cpp
    src/data/repositories/execution_unit_repository.cpp
    src/data/repositories/inspection_type_repository.cpp
    src/data/repositories/item_repository.cpp
    src/data/repositories/master_video_repository.cpp
    src/data/repositories/project_repository.cpp
    src/data/repositories/result_repository.cpp
    src/data/repositories/result_image_repository.cpp
    src/data/repositories/session_item_repository.cpp
    src/data/repositories/session_repository.cpp
    src/data/repositories/timeline_thumbnail_repository.cpp
    src/data/repositories/tooling_repository.cpp
    src/data/repositories/video_clip_repository.cpp
)

target_include_directories(travis_data_database
    PUBLIC
        ${CMAKE_CURRENT_SOURCE_DIR}/include
)

target_link_libraries(travis_data_database
    PUBLIC
        Qt6::Core
        Qt6::Sql
)

add_library(travis_services
    src/services/execution_service.cpp
    src/services/inspection_clip_service.cpp
    src/services/project_service.cpp
    src/services/result_media_service.cpp
    src/services/result_service.cpp
    src/services/session_service.cpp
    src/services/structure_service.cpp
    src/services/video_service.cpp
)

target_include_directories(travis_services
    PUBLIC
        ${CMAKE_CURRENT_SOURCE_DIR}/include
)

target_link_libraries(travis_services
    PUBLIC
        Qt6::Core
        travis_data_database
)

if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
