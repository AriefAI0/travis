cmake_minimum_required(VERSION 3.21)

project(travis VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)

include(CTest)

find_package(Qt6 REQUIRED COMPONENTS Core Quick Qml Sql Test)

qt_standard_project_setup()
qt_policy(SET QTP0001 NEW)
qt_policy(SET QTP0004 NEW)

set(TRAVIS_GSTREAMER_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/runtime/gstreamer/1.0/msvc_x86_64)

add_library(travis_data_database
    src/data/database/application_identity.cpp
    src/data/database/database_bootstrap.cpp
    src/data/database/database_manager.cpp
    src/data/database/database_paths.cpp
    src/data/repositories/asset_repository.cpp
    src/data/repositories/base_repository.cpp
    src/data/repositories/component_repository.cpp
    src/data/repositories/execution_unit_repository.cpp
    src/data/repositories/inspection_type_repository.cpp
    src/data/repositories/item_repository.cpp
    src/data/repositories/master_video_repository.cpp
    src/data/repositories/project_repository.cpp
    src/data/repositories/result_repository.cpp
    src/data/repositories/result_image_repository.cpp
    src/data/repositories/session_item_repository.cpp
    src/data/repositories/session_repository.cpp
    src/data/repositories/timeline_thumbnail_repository.cpp
    src/data/repositories/tooling_repository.cpp
    src/data/repositories/video_clip_repository.cpp
)

target_include_directories(travis_data_database
    PUBLIC
        ${CMAKE_CURRENT_SOURCE_DIR}/include
)

target_link_libraries(travis_data_database
    PUBLIC
        Qt6::Core
        Qt6::Sql
)

add_library(travis_services
    src/services/execution_service.cpp
    src/services/inspection_clip_service.cpp
    src/services/project_service.cpp
    src/services/result_media_service.cpp
    src/services/result_service.cpp
    src/services/session_service.cpp
    src/services/structure_service.cpp
    src/services/video_service.cpp
)

target_include_directories(travis_services
    PUBLIC
        ${CMAKE_CURRENT_SOURCE_DIR}/include
)

target_link_libraries(travis_services
    PUBLIC
        Qt6::Core
        travis_data_database
)

add_library(travis_application
    include/application/playback/playback_workflow_service.h
    include/application/recording/recording_workflow_service.h
    include/application/recovery/recording_recovery_service.h
    src/app/app_context.cpp
    src/application/playback/playback_workflow_service.cpp
    src/application/recording/recording_workflow_service.cpp
    src/application/recovery/recording_recovery_service.cpp
)

target_include_directories(travis_application
    PUBLIC
        ${CMAKE_CURRENT_SOURCE_DIR}/include
)

target_link_libraries(travis_application
    PUBLIC
        Qt6::Core
        travis_services
        travis_media_engine
)

add_library(travis_media_engine
    src/mediaEngine/core/media_runtime.cpp
    src/mediaEngine/core/video_sink_selector.cpp
    src/mediaEngine/discovery/source_discovery.cpp
    src/mediaEngine/metering/audio_meter_engine.cpp
    src/mediaEngine/pipeline/caps.cpp
    src/mediaEngine/pipeline/dynamic_pad.cpp
    src/mediaEngine/pipeline/errors.cpp
    src/mediaEngine/preview/preview_engine.cpp
    src/mediaEngine/recording/recording_engine.cpp
    src/mediaEngine/recording/pipeline/av_recording_pipeline.cpp
    src/mediaEngine/recording/shared/recording_finalization.cpp
    src/mediaEngine/recording/shared/recording_validation.cpp
    src/mediaEngine/session/media_source_session.cpp
    src/mediaEngine/sources/device_capture/device_capture_video_source.cpp
    src/mediaEngine/sources/ndi/ndi_source_reference.cpp
    src/mediaEngine/sources/ndi/ndi_video_source.cpp
)

target_include_directories(travis_media_engine
    PUBLIC
        ${CMAKE_CURRENT_SOURCE_DIR}/include
        ${TRAVIS_GSTREAMER_ROOT}/include
        ${TRAVIS_GSTREAMER_ROOT}/include/gstreamer-1.0
        ${TRAVIS_GSTREAMER_ROOT}/include/glib-2.0
        ${TRAVIS_GSTREAMER_ROOT}/lib/glib-2.0/include
)

target_link_directories(travis_media_engine
    PUBLIC
        ${TRAVIS_GSTREAMER_ROOT}/lib
)

target_link_libraries(travis_media_engine
    PUBLIC
        Qt6::Core
        Qt6::Quick
        gstreamer-1.0
        gstvideo-1.0
        gstapp-1.0
        gobject-2.0
        glib-2.0
)

add_library(travis_ui
    include/ui/controllers/preview_surface_controller.h
    include/ui/viewmodels/audio_meter_viewmodel.h
    include/ui/viewmodels/playback_viewmodel.h
    include/ui/viewmodels/recording_viewmodel.h
    include/ui/viewmodels/source_discovery_viewmodel.h
    src/ui/controllers/preview_surface_controller.cpp
    src/ui/viewmodels/audio_meter_viewmodel.cpp
    src/ui/viewmodels/playback_viewmodel.cpp
    src/ui/viewmodels/recording_viewmodel.cpp
    src/ui/viewmodels/source_discovery_viewmodel.cpp
)

target_include_directories(travis_ui
    PUBLIC
        ${CMAKE_CURRENT_SOURCE_DIR}/include
)

target_link_libraries(travis_ui
    PUBLIC
        Qt6::Core
        Qt6::Qml
        Qt6::Quick
        travis_application
        travis_media_engine
)

qt_add_executable(travis_app
    src/app/application_bootstrap.cpp
    src/app/main.cpp
)

qt_add_qml_module(travis_app
    URI Travis
    VERSION 1.0
    QML_FILES
        qml/Main.qml
)

target_include_directories(travis_app
    PRIVATE
        ${CMAKE_CURRENT_SOURCE_DIR}/include
)

target_link_libraries(travis_app
    PRIVATE
        Qt6::Core
        Qt6::Qml
        Qt6::Quick
        travis_data_database
        travis_ui
)

if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
